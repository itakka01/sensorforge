###########################
# AUTHENTICATION        
###########################


web_auth_enabled=1
web_username=ja1
# passwort min 8  max 62 characters!
web_password=sensor4337


############################
# KAMERA
############################

# Kamera-Modell:
#   OV2640
#   OV3660
#   OV5640
camera=OV3660

# Auflösung:
#   160x120
#   320x240
#   640x480
#   800x600
#   1024x768
#   1280x1024
#   1600x1200
#   2048x1536
resolution=1024x768

# Bildrotation:
# 0
# 180
rotation=0



# Bilder pro Sekunde
fps=5

# JPEG-Qualität:
# 0 = beste Qualität
# 63 = schlechteste Qualität
quality=12


# Belichtungsautomatik:
# 1 = Auto Exposure aktiv
# 0 = Auto Exposure deaktiviert
camera_auto_exposure=1

# Zielhelligkeit der Auto Exposure:
# Bereich -2..2
# Negativer = dunkleres Belichtungsziel; kann Bewegungsunschärfe reduzieren
# Startwert für Bewegung: -1
camera_ae_level=-2




############################
# AUFNAHME / TRIGGER
############################

# Recording-Container:
#   avi = MJPEG AVI + separate standard .srt subtitle file
#   mkv = MJPEG in Matroska + embedded UTF-8 subtitle track
#
# Hinweis: MKV wird in einem späteren Implementierungsschritt aktiviert.
recording_format=avi

# Datum/Uhrzeit als Untertitel erzeugen:
#   1 = aktiviert
#   0 = deaktiviert
timestamp_enabled=1

# Nachlaufzeit nach letzter Bewegung (ms)
post_record_ms=3000



# Maximale Dauer eines einzelnen Aufnahme-Segments in Sekunden.
# 0 = keine zeitliche Segmentbegrenzung
recording_segment_seconds=30

# Maximale Groesse eines einzelnen Video-Segments in MB.
# 0 = keine zusaetzliche Groessenbegrenzung
recording_segment_max_mb=20





############################
# SLEEP / STROMSPAREN
############################

# Sleep-Modus bei Inaktivitaet und ausgeschaltetem WLAN:
#   off         = kein Sleep, ESP32 bleibt wach
#   light_sleep = schneller Wake, Programmzustand bleibt erhalten
#   deep_sleep  = maximal stromsparend, Wake fuehrt zu einem Neustart
sleep_mode=light_sleep

# Wartezeit nach Aufnahmeende / letzter Aktivitaet vor Sleep (ms)
# Bereich: 0..60000
# Standard: 2000
sleep_delay_ms=2000




############################
# LED
############################

# 1 = LED während Aufnahme
# 0 = LED aus
led_enabled=1

############################
# SPEICHER / SD-SICHERHEIT
############################

# Mindestreserve auf der SD-Karte in MB.
# Standard: 100 MB
min_free_space_mb=100

# Verhalten beim Erreichen der Mindestreserve:
# rollover = älteste vollständige Aufnahmen löschen
# stop     = keine weitere Aufnahme, bis wieder genug Platz frei ist
disk_full_action=rollover




############################
# WLAN / NTP
############################

# WLAN-Verhalten beim Systemstart:
# off             = WLAN beim Systemstart nicht einschalten
# on              = WLAN beim Systemstart einschalten
# on_missing_time = WLAN beim Systemstart nur einschalten,
#                   wenn keine gueltige Uhrzeit vorhanden ist
wifi_on_system_start=on

# Automatische WLAN-Abschaltung nach Inaktivität in Sekunden.
# 0 = WLAN wird nicht automatisch/manuell abgeschaltet
# Standard: 60 Sekunden
wifi_timeout_sec=60

hotspot_enabled=1
hotspot_password=sensor4337
hotspot_hidden=0



############################
# ZEITZONE
############################

# Zeitzone im POSIX-TZ-Format.
#
# Österreich / Mitteleuropa:
#   Winterzeit = CET  (UTC+1)
#   Sommerzeit = CEST (UTC+2)
timezone=CET-1CEST,M3.5.0,M10.5.0/3
